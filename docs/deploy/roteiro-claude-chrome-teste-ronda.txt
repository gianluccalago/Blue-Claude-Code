insert into public.nfc_tags (uid, quarto_id, last_counter, observacao, cadastrada_por)
select '04000000000001', q.id, 0, 'TESTE - apagar', 'teste'
  from public.quarto q
 where q.modulo = 5
   and not exists (select 1 from public.residentes r join public.leito l on l.id = r.leito_id
                    where l.quarto_id = q.id and r.status_hospede = 'ativo')
   and not exists (select 1 from public.chamado c where c.quarto_id = q.id and c.status = 'aberto')
 order by q.codigo limit 1
on conflict (uid) do nothing;

(async () => {
  const chave = Object.keys(localStorage).find((k) => /^sb-.*-auth-token$/.test(k));
  const sessao = chave ? JSON.parse(localStorage.getItem(chave)) : null;
  const tablet = localStorage.getItem("blue-device-token");
  if (!sessao?.access_token) return console.log("Sem sessão: faça login no app nesta aba.");
  if (!tablet) return console.log("Sem tablet: faça a PARTE 2 neste navegador.");
  const chamar = async (contador) => {
    const r = await fetch("https://hqfrxdwumyyujewjgxgc.supabase.co/functions/v1/verify-round", {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${sessao.access_token}`, "x-device-token": tablet },
      body: JSON.stringify({ url: `https://app.blueseniorliving.com.br/r?m=04000000000001x${contador}`, serial_number: "04:00:00:00:00:00:01" }),
    });
    return `${r.status} ${await r.text()}`;
  };
  console.log("A) primeira leitura:", await chamar("000001"));
  console.log("B) mesma leitura de novo:", await chamar("000001"));
  console.log("C) pulando o contador:", await chamar("000004"));
})();

B) 200 com "ok":false e "status":"rejeitada_contador_repetido".
C) 200 com "ok":true, "status":"valida" e a sinalização "leituras_nao_registradas".
